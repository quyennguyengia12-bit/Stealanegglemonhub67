-- Safe descendant iteration (prevents stack overflow)
local function forEachDescendant(root, callback, maxDepth)
    maxDepth = maxDepth or 10
    local function recurse(node, depth)
        if depth > maxDepth then return end
        local children = node:GetChildren()
        for i = 1, #children do
            local child = children[i]
            local ok, err = pcall(callback, child)
            if not ok then
                warn(string_format("[forEachDescendant] Error: %s", tostring(err)))
            end
            recurse(child, depth + 1)
        end
    end
    recurse(root, 0)
end

-- Connection tracking system
local function trackConnection(conn, tag)
    tag = tag or "default"
    if not connectionRegistry[tag] then
        connectionRegistry[tag] = {}
    end
    table_insert(connectionRegistry[tag], conn)
    return conn
end

-- Cleanup all connections
local function cleanupConnections(tag)
    if connectionRegistry[tag] then
        for i = #connectionRegistry[tag], 1, -1 do
            local conn = connectionRegistry[tag][i]
            if conn and conn.Disconnect then
                pcall(function() conn:Disconnect() end)
            end
            table_remove(connectionRegistry[tag], i)
        end
    end
end

-- Safe parent function
local function SafeParent(gui)
    local okH, h = pcall(function() return gethui and gethui() end)
    if okH and h and typeof(h) == "Instance" and h.Parent then
        pcall(function() gui.Parent = h end)
        if gui.Parent == h then return end
    end
    pcall(function() gui.Parent = CoreGui end)
    if not gui.Parent then 
        pcall(function() gui.Parent = LocalPlayer:WaitForChild("PlayerGui", 5) end) 
    end
end

-- Enhanced instance creator
local function mk(cls, parent, props)
    local i = Instance.new(cls)
    for k, v in pairs(props) do 
        local ok, err = pcall(function() i[k] = v end)
        if not ok then
            warn(string_format("[mk] Failed to set %s.%s: %s", cls, tostring(k), tostring(err)))
        end
    end
    if parent then 
        pcall(function() i.Parent = parent end)
    end
    return i
end

local function WalkTo(pos, timeout)
    if not Hum or not HRP or not pos then return false end
    
    pcall(function() Hum.WalkSpeed = State.FarmSpeed end)
    pcall(function() Hum:MoveTo(pos) end)
    
    local start = os_clock()
    local lastIssue = start
    local lastPos = HRP.Position
    local stuckAt = start
    local jumps = 0
    timeout = timeout or 30
    
    while os_clock() - start < timeout do
        if State.FarmMode == "off" then return false end
        if not HRP or not Hum or Hum.Health <= 0 then return false end
        
        NoteGoodPos()
        if RewindIfFell() then return false end
        if (HRP.Position - pos).Magnitude < 6 then return true end
        
        if (HRP.Position - lastPos).Magnitude < 0.5 then
            if os_clock() - stuckAt > 2 then
                jumps = jumps + 1
                pcall(function() Hum.Jump = true end)
                pcall(function() Hum:MoveTo(pos) end)
                stuckAt = os_clock()
                if jumps > 3 then return false end
            end
        else
            lastPos = HRP.Position
            stuckAt = os_clock()
        end
        
        if os_clock() - lastIssue > 2 then 
            pcall(function() Hum:MoveTo(pos) end)
            lastIssue = os_clock() 
        end
        
        task_wait(0.25)
    end
    
    return HRP and (HRP.Position - pos).Magnitude < 8
end

local function hasKW(s, list)
    if not s then return false end
    s = string_lower(s)
    for i = 1, #list do 
        if string_find(s, list[i], 1, true) then return true end 
    end
    return false
end

local function hasExplicitPlace(text)
    text = string_lower(text)
    for i = 1, #EXPLICIT_PLACE_PATTERNS do
        if string_find(text, EXPLICIT_PLACE_PATTERNS[i], 1, true) then return true end
    end
    return false
end

-- Enhanced PromptPos with safety limits
local function PromptPos(p)
    local par = p.Parent
    if not par then return nil end
    if par:IsA("BasePart") then return par.Position end
    if par:IsA("Model") and par.PrimaryPart then return par.PrimaryPart.Position end
    
    -- BFS with depth limit
    local queue = {{par, 0}}
    local head = 1
    while head <= #queue do
        local current, depth = queue[head][1], queue[head][2]
        head = head + 1
        
        if depth > 10 then break end -- Prevent infinite loops
        
        if current:IsA("BasePart") then return current.Position end
        
        local children = current:GetChildren()
        for i = 1, #children do
            if #queue < 200 then -- Memory safety limit
                table_insert(queue, {children[i], depth + 1})
            end
        end
    end
    return nil
end

local function ScanInto(root)
    local n0 = #State.Cache.Steal + #State.Cache.Place
    pcall(function()
        forEachDescendant(root, function(obj)
            if not obj or not obj.Parent then return end -- Guard against destroyed objects
            
            if obj:IsA("ProximityPrompt") and obj.Enabled then
                local text = string_format("%s %s %s", 
                    obj.ActionText or "", 
                    obj.ObjectText or "", 
                    obj.Name)
                local pos = PromptPos(obj)
                
                if pos then
                    if hasExplicitPlace(text) then
                        table_insert(State.Cache.Place, { p = obj, pos = pos })
                    elseif hasKW(text, PLACE_KW) and not hasKW(text, STEAL_KW) then
                        table_insert(State.Cache.Place, { p = obj, pos = pos })
                    elseif hasKW(text, STEAL_KW) then
                        table_insert(State.Cache.Steal, { p = obj, pos = pos })
                    end
                end
            end
        end, 8)
    end)
    return (#State.Cache.Steal + #State.Cache.Place) > n0
end

local function UpdateCache()
    State.Cache.Steal, State.Cache.Place = {}, {}
    ScanInto(Workspace)
end

local function PruneSkip()
    local now = os_clock()
    local toRemove = {}
    
    for k, v in pairs(SkipMap) do
        local alive = true
        pcall(function() alive = (k.Parent ~= nil) end)
        if v < now or not alive then 
            table_insert(toRemove, k)
        end
    end
    
    for i = 1, #toRemove do
        SkipMap[toRemove[i]] = nil
    end
end

local function MaybeScan(force)
    if force or os_clock() - LastScan > 5 then
        LastScan = os_clock()
        PruneSkip()
        UpdateCache()
    end
end

                    CarrySrc = "nút GUI: " .. tostring(g.Text)

                            break 
                        end 
                    end

                        local keywords = { 
                            "thả trứng", "drop egg", "đặt trứng", 
                            "deliver egg", "giao trứng" 
                        }
                        for i = 1, #keywords do 
                            if string_find(t, keywords[i], 1, true) then 
                                CarrySrc = "label: " .. tostring(g.Text)
                                found = true
                                return

                            if string_find(t, CARRY_EXTRA[i], 1, true) then 
                                CarrySrc = "label: " .. tostring(g.Text)
                                found = true
                                return 
                            end 
                        end
                    end
                end
            end
        end, 6)
        
        if found then return true end
    end
    
    if Char then
        local foundModel = false
        forEachDescendant(Char, function(c)
            if foundModel then return end
            local n = string_lower(c.Name)
            if (string_find(n, "egg", 1, true) or string_find(n, "trứng", 1, true)) and 
               not string_find(n, "bat", 1, true) and 
               not string_find(n, "gậy", 1, true) and 
               (c:IsA("BasePart") or c:IsA("Model") or c:IsA("MeshPart")) then
                CarrySrc = "model: " .. c.Name
                foundModel = true
            end
        end, 5)
        
        if foundModel then return true end
    end
    
    CarrySrc = ""
    return false
end

local function TrySteal(prompt, pos)
    if not prompt or not HRP or not prompt.Parent then return false end
    
    local maxD = (prompt.MaxActivationDistance or 10)
    for attempt = 1, 3 do
        if not prompt.Parent or not prompt.Enabled then return true end
        if IsCarrying() then return true end
        
        if (HRP.Position - pos).Magnitude > maxD then
            WalkTo(pos, 6)
            task_wait(0.2)
        end
        
        local okB = pcall(function() prompt:InputHoldBegin() end)
        if okB then
            local hold = math_max(0.1, math_max(prompt.HoldDuration or 0, 3))
            local t0 = os_clock()
            local result = nil
            
            while os_clock() - t0 < hold do
                task_wait(0.08)
                if not prompt.Parent or not prompt.Enabled then 
                    result = "taken"
                    break 
                end
                if not HRP then 
                    result = "dead"
                    break 
                end
                if IsCarrying() then 
                    result = "carry"
                    break 
                end
            end
            
            pcall(function() prompt:InputHoldEnd() end)
            
            if result == "taken" or result == "carry" then return true end
            if result == "dead" then return false end
            
            task_wait(0.15)
            if IsCarrying() then return true end
            if not prompt.Parent or not prompt.Enabled then return true end
            
            pcall(function() prompt:InputHoldBegin() end)
            task_wait(0.12)
            pcall(function() prompt:InputHoldEnd() end)
            task_wait(0.15)
            
            if IsCarrying() then return true end
            if not prompt.Parent or not prompt.Enabled then return true end
        end
        task_wait(0.25)
    end
    return IsCarrying()
end

end

local function RobustReturnToNest(targetPos, maxAttempts)
    maxAttempts = maxAttempts or 5
    for attempt = 1, maxAttempts do
        if State.FarmMode == "off" then return false end
        if not IsCarrying() then return true end
        if not HRP or not Hum or Hum.Health <= 0 then return false end
        if (HRP.Position - targetPos).Magnitude < 12 then return true end
        
        local jumpAlive = true
        task_spawn(function()
            while jumpAlive and scriptAlive and Hum and Hum.Health > 0 and State.FarmMode ~= "off" do
                pcall(function() Hum.Jump = true end)
                task_wait(0.7)
            end
        end)
        
        local dist = (HRP.Position - targetPos).Magnitude
        local dynamicTimeout = math_max(30, dist / math_max(16, State.FarmSpeed) * 2.5)
        local ok = WalkTo(targetPos, dynamicTimeout)
        jumpAlive = false
        
        if ok and HRP and (HRP.Position - targetPos).Magnitude < 12 then return true end
        SetStatus(string_format("🔄 Thử lại #%d/%d", attempt, maxAttempts))
        task_wait(0.8)
        MaybeScan(true)
    end
    return false
end

local CarryDebugged = false
local function StartFarm()
    if State.FarmRunning then return end
    State.FarmRunning = true
    task_spawn(function()
        MaybeScan(true)
        if not CarryDebugged then
            CarryDebugged = true
            print(string_format("[V48] 🚀 Bắt đầu farm - Cache: %d trứng, %d ổ", 
                #State.Cache.Steal, #State.Cache.Place))
            if IsCarrying() then NotifyRef("warn", "🔍 CARRY SRC:", CarrySrc, 6) end
        end
        
        while State.FarmMode ~= "off" do
            local ok, err = pcall(function()
                UpdateCharacter()
                if not HRP or not Hum or Hum.Health <= 0 then task_wait(1) return end
                MaybeScan(false)

                if IsCarrying() then
                    local dp, src = State.SafeNest, "nest"
                    if not dp then
                        MaybeScan(true)
                        local bp, bd = nil, math_huge
                        local cache = State.Cache.Place
                        for i = 1, #cache do
                            local d = cache[i]
                            if d.p and d.p.Parent and d.pos then
                                local dist = (HRP.Position - d.pos).Magnitude
                                if dist < bd then bd, bp = dist, d end
                            end
                        end
                        if bp then dp, src = bp.pos, "prompt" end
                    end
                    if not dp and State.SafeZone then dp, src = State.SafeZone, "safezone" end
                    if not dp then
                        if os_clock() - LastRemind > 15 then
                            LastRemind = os_clock()
                            NotifyRef("warn", "CHƯA CÓ Ổ!", "Đứng ở vùng an toàn → bấm 🏠 Lưu", 5)
                        end
                        SetStatus("⛔ Cần 🏠 Ổ AN TOÀN!")
                        task_wait(2)
                        return
                    end
                    if State.Ghost and PlayerNearby(10) then task_wait(2) return end
                    SetStatus(string_format("🏠 Về ổ (%s)...", src))

                    if RobustReturnToNest(dp, 5) then
                        local t0 = os_clock()
                        while os_clock() - t0 < 3 and IsCarrying() do task_wait(0.4) end
                        MaybeScan(false)
                        if IsCarrying() then FirePlacePrompt(14) end
                        if IsCarrying() then
                            local offsets = { 
                                Vector3_new(6, 0, 0), Vector3_new(-6, 0, 0), 
                                Vector3_new(0, 0, 6), Vector3_new(0, 0, -6) 
                            }
                            for i = 1, #offsets do
                                if not IsCarrying() then break end
                                WalkTo(dp + offsets[i], 4)
                                task_wait(0.6)
                            end
                            if IsCarrying() then FirePlacePrompt(20) end
                            local t1 = os_clock()
                            while os_clock() - t1 < 4 and IsCarrying() do task_wait(0.4) end
                        end
                        if not IsCarrying() then
                            AssumeCarryUntil = 0
                            State.EggsPlaced = State.EggsPlaced + 1
                            State.DeliverFails = 0
                            if not State.SafeNest then
                                State.SafeNest = dp
                                NotifyRef("success", "🏠 AUTO NEST", "Tự lưu ổ giao thành công!", 4)
                            end
                            SetStatus(string_format("✅ Giao #%d", State.EggsPlaced))
                            NotifyRef("success", "🏠 GIAO OK!", string_format("Tổng: %d", State.EggsPlaced), 3)
                        else
                            State.DeliverFails = State.DeliverFails + 1
                            SetStatus(string_format("⚠️ Chưa giao x%d", State.DeliverFails))
                            if State.DeliverFails >= 3 then
                                if src == "nest" then
                                    State.SafeNest = nil
                                    NotifyRef("warn", "🧹 Ổ SAI!", "Sẽ thử đích safezone lần sau", 4)
                                end
                                State.DeliverFails = 0
                            end
                        end
                        HumanPause()
                    end
                else
                    if State.Ghost and PlayerNearby(10) then task_wait(2) return end
                    local now = os_clock()
                    local best, bestDist = nil, math_huge
                    local cache = State.Cache.Steal
                    
                    if State.SpawnNest then
                        for i = 1, #cache do
                            local d = cache[i]
                            if d.p and d.p.Parent and d.pos and d.pos.Y > -20 and 
                               not (SkipMap[d.p] and now < SkipMap[d.p]) then
                                local fromSpawn = (State.SpawnNest - d.pos).Magnitude
                                local fromMe = (HRP.Position - d.pos).Magnitude
                                if fromSpawn < 80 and fromMe < State.Radius and fromMe < bestDist then 
                                    bestDist, best = fromMe, d 
                                end
                            end
                        end
                    end
                    
                    if not best then
                        for i = 1, #cache do
                            local d = cache[i]
                            if d.p and d.p.Parent and d.pos and d.pos.Y > -20 and 
                               not (SkipMap[d.p] and now < SkipMap[d.p]) then
                                local dist = (HRP.Position - d.pos).Magnitude
                                if dist < State.Radius and dist < bestDist then 
                                    bestDist, best = dist, d 
                                end
                            end
                        end
                    end
                    
                    if not best and State.SpawnNest and (HRP.Position - State.SpawnNest).Magnitude > 15 then
                        SetStatus("🥚 Về Ổ LẤY TRỨNG chờ...")
                        WalkTo(State.SpawnNest, 25)
                        MaybeScan(true)
                        task_wait(1.5)
                        return
                    end
                    
                    if best then
                        SetStatus(string_format("🚶 Tới trứng (%dm)", math_floor(bestDist)))
                        if WalkTo(best.pos, 25) then
                            task_wait(math_random(3, 8) / 10)
                            if TrySteal(best.p, best.pos) then
                                AssumeCarryUntil = os_clock() + 45
                                State.EggsStolen = State.EggsStolen + 1
                                SetStatus(string_format("✅ Cướp #%d", State.EggsStolen))
                                NotifyRef("success", "🥚 CƯỚP OK!", string_format("Tổng: %d", State.EggsStolen), 3)
                            else
                                SkipMap[best.p] = os_clock() + 10
                                SetStatus("⚠️ Hụt, đổi mục tiêu...")
                            end
                            HumanPause()
                        else
                            SkipMap[best.p] = os_clock() + 25
                            SetStatus("⚠️ Kẹt đường, đổi trứng...")
                        end
                    else
                        SetStatus("⏳ Quét lại trứng...")
                        MaybeScan(true)
                        task_wait(2.5)
                    end
                end
            end)
            if not ok then warn(string_format("[V48 FARM ERR] %s", tostring(err))) end
            task_wait(0.2)
        end
        State.FarmRunning = false
        pcall(function() if Hum then Hum.WalkSpeed = 16 end end)
        SetStatus("🟢 Sẵn sàng")
    end)
end
